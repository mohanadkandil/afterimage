cmake_minimum_required(VERSION 3.20)
project(Litt VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
add_library(litt_core core/store.cpp)
target_include_directories(litt_core PUBLIC core vendor ${SQLite3_INCLUDE_DIRS})
target_link_libraries(litt_core PUBLIC ${SQLite3_LIBRARIES})
target_compile_options(litt_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
 add_executable(core_tests tests/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE litt_core)
 add_test(NAME core COMMAND core_tests)
 set_tests_properties(core PROPERTIES TIMEOUT 30)
endif()
if(APPLE)
 enable_language(OBJCXX)
 set(CMAKE_OBJCXX_STANDARD 20)
 set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
 add_executable(Litt MACOSX_BUNDLE native/main.mm)
 target_link_libraries(Litt PRIVATE litt_core "-framework Cocoa" "-framework WebKit" "-framework ScreenCaptureKit" "-framework Vision" "-framework ImageIO" "-framework UniformTypeIdentifiers")
 target_compile_options(Litt PRIVATE -fobjc-arc -Wall -Wextra)
 set_target_properties(Litt PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/native/Info.plist")
 if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME native_ocr_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:Litt>)
  set_tests_properties(native_ocr_cli PROPERTIES TIMEOUT 90)
 endif()
 add_custom_command(TARGET Litt POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/web" "$<TARGET_BUNDLE_DIR:Litt>/Contents/Resources/web" COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/assets/AppIcon.icns" "$<TARGET_BUNDLE_DIR:Litt>/Contents/Resources/AppIcon.icns")
endif()
