cmake_minimum_required(VERSION 3.20)
project(Afterimage VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
add_library(afterimage_core core/store.cpp)
target_include_directories(afterimage_core PUBLIC core vendor ${SQLite3_INCLUDE_DIRS})
target_link_libraries(afterimage_core PUBLIC ${SQLite3_LIBRARIES})
target_compile_options(afterimage_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
 add_executable(core_tests tests/core_tests.cpp)
 target_link_libraries(core_tests PRIVATE afterimage_core)
 add_test(NAME core COMMAND core_tests)
 set_tests_properties(core PROPERTIES TIMEOUT 30)
endif()
if(APPLE)
 enable_language(OBJCXX)
 set(CMAKE_OBJCXX_STANDARD 20)
 set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0")
 find_program(SWIFTC swiftc REQUIRED)
 file(GLOB_RECURSE SWIFT_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/native/SwiftUI/*.swift")
 set(SWIFT_LIBRARY "${CMAKE_BINARY_DIR}/libAfterimageViews.dylib")
 add_custom_command(OUTPUT "${SWIFT_LIBRARY}" "${CMAKE_BINARY_DIR}/AfterimageViews-Swift.h"
   COMMAND ${SWIFTC} -sdk "${CMAKE_OSX_SYSROOT}" -swift-version 6 -O -parse-as-library -emit-library
     -module-name AfterimageViews -emit-objc-header -emit-objc-header-path "${CMAKE_BINARY_DIR}/AfterimageViews-Swift.h"
     -target ${CMAKE_SYSTEM_PROCESSOR}-apple-macosx14.0 ${SWIFT_SOURCES}
     -Xlinker -install_name -Xlinker @rpath/libAfterimageViews.dylib -o "${SWIFT_LIBRARY}"
   DEPENDS ${SWIFT_SOURCES} VERBATIM)
 add_custom_target(afterimage_swiftui DEPENDS "${SWIFT_LIBRARY}")
 add_executable(Afterimage MACOSX_BUNDLE native/main.mm native/NativeUI.mm native/SegmentCodec.mm)
 add_dependencies(Afterimage afterimage_swiftui)
 target_include_directories(Afterimage PRIVATE "${CMAKE_BINARY_DIR}")
 set_target_properties(Afterimage PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks")
 target_link_libraries(Afterimage PRIVATE "${SWIFT_LIBRARY}" afterimage_core "-framework Cocoa" "-framework QuartzCore" "-framework ScreenCaptureKit" "-framework Vision" "-framework ImageIO" "-framework AVFoundation" "-framework CoreMedia" "-framework CoreVideo" "-framework UniformTypeIdentifiers")
 target_compile_options(Afterimage PRIVATE -fobjc-arc -Wall -Wextra)
 set_target_properties(Afterimage PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/native/Info.plist")
 if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME native_ocr_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:Afterimage>)
  add_test(NAME storage_optimization COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/storage_optimization.py $<TARGET_FILE:Afterimage>)
  add_test(NAME video_storage COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/video_storage.py $<TARGET_FILE:Afterimage>)
  set_tests_properties(native_ocr_cli storage_optimization video_storage PROPERTIES TIMEOUT 90)
 endif()
 add_custom_command(TARGET Afterimage POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_DIR:Afterimage>/Contents/Frameworks" COMMAND ${CMAKE_COMMAND} -E copy "${SWIFT_LIBRARY}" "$<TARGET_BUNDLE_DIR:Afterimage>/Contents/Frameworks/libAfterimageViews.dylib" COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/assets/AppIcon.icns" "$<TARGET_BUNDLE_DIR:Afterimage>/Contents/Resources/AppIcon.icns")
endif()
